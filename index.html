CC102 Game

Hangman Game

This is a console-based word guessing game developed using Java. The player needs to guess the hidden word by entering one letter at a time. The player has five attempts to guess the correct word. The game also includes a score and replay option. 

import java.util.Scanner;

public class HangmanGame {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        String[] words = {
            "COMPUTER",
            "PROGRAMMING",
            "INTERNET",
            "INFORMATION TECHNOLOGY",
            "SOFTWARE",
            "HARDWARE",
            "DATABASE",
            "ALGORITHM",
            "DEBUGGING"
        };

        int score = 0;
        String playAgain;

        System.out.println("HANGMAN GAME");

        do {

            String word = words[(int)(Math.random() * words.length)];
            char[] guessed = new char[word.length()];

            for (int i = 0; i < guessed.length; i++) {
                guessed[i] = '_';
            }

            int attempts = 5;
            boolean won = false;

            System.out.println("\nGuess the hidden word!");

            while (attempts > 0 && !won) {

                System.out.print("\nWord: ");

                for (char letter : guessed) {
                    System.out.print(letter + " ");
                }

                System.out.println("\nAttempts left: " + attempts);
                System.out.print("Enter a letter: ");

                char guess = input.next().toUpperCase().charAt(0);
                boolean correct = false;

                for (int i = 0; i < word.length(); i++) {

                    if (word.charAt(i) == guess && guessed[i] == '_') {
                        guessed[i] = guess;
                        correct = true;
                    }

                }

                if (correct) {
                    System.out.println("Correct letter!");
                } else {
                    System.out.println("Wrong letter!");
                    attempts--;
                }

                won = true;

                for (char letter : guessed) {

                    if (letter == '_') {
                        won = false;
                        break;
                    }

                }

            }

            if (won) {

                System.out.println("\nCongratulations!");
                System.out.println("You guessed the word: " + word);
                score++;

            } else {

                System.out.println("\nGame Over!");
                System.out.println("The correct word was: " + word);

            }

            System.out.println("Your Score: " + score);

            System.out.print("\nPlay again? (yes/no): ");
            playAgain = input.next();

        } while (playAgain.equalsIgnoreCase("yes"));

        System.out.println("THANK YOU FOR PLAYING!");
        System.out.println("Final Score: " + score);

        input.close();

    }

}
